-- 3) Which volunteer(s) are working on transporting tasks? Query should retrieve names rather than ids.
-- Q3 
SELECT volunteer.volunteerName
FROM volunteer
	INNER JOIN assignment
		ON volunteer.volunteerId= assignment.volunteerId
    INNER JOIN task
		ON assignment.taskCode =task.taskCode
	INNER JOIN task_type
		ON task.taskTypeId = task_type.taskTypeId
WHERE task_type.taskTypeName= 'transporting';

-- 5)Which type(s) of package contain some kind of bottle? 
-- Q5
SELECT DISTINCT PT.packageTypeName
FROM item 
  INNER JOIN package_contents AS PC
    ON item.itemId = PC.itemId
  INNER JOIN package AS P
    ON PC.packageId = P.packageId
  INNER JOIN package_type AS PT
    ON P.packageTypeId = PT.packageTypeId
WHERE item.itemDescription LIKE '%bottle%';

-- 7) Which task(s) are assigned to volunteer(s) that live in New Jersey (NJ)? Answer should have the task description and not the task ids.
 -- Q7
SELECT DISTINCT task.taskDescription
FROM volunteer AS V
  INNER JOIN assignment AS assign
    ON V.volunteerId = assign.volunteerId
  INNER JOIN task 
    ON assign.taskCode = task.taskCode
WHERE V.volunteerAddress LIKE '% NJ%';

-- 'Prepare 5,000 packages'

-- 9) Which volunteers have been assigned to tasks that include packing spam? Answer should have the volunteer names and not their ids.
-- Q9 
SELECT DISTINCT V.volunteerName
FROM volunteer AS V
  INNER JOIN assignment AS assign
    ON V.volunteerId = assign.volunteerId
  INNER JOIN task 
    ON assign.taskCode = task.taskCode
  INNER JOIN package AS P
    ON task.taskCode = P.taskCode
  INNER JOIN package_contents AS PC
    ON P.packageId = PC.packageId
  INNER JOIN item 
    ON PC.itemId = item.itemId
WHERE item.itemDescription = 'can of spam';

-- 10) Which item(s) (if any) have a total value of exactly $100 in one package? Answer should be item descriptions.
-- Q10

SELECT item.itemDescription
FROM item 
  INNER JOIN package_contents AS PC
    ON item.itemId = PC.itemId
WHERE (item.itemValue * PC.itemQuantity) = 100.0;

-- 11) How many volunteers are assigned to tasks with each different status? The answer lowest)
-- Q11

SELECT TS.taskStatusName,
  COUNT(assign.volunteerId) AS VolunteerCount
FROM
  assignment AS assign
  INNER JOIN task 
    ON assign.taskCode = task.taskCode
  INNER JOIN task_status AS TS
    ON task.taskStatusId = TS.taskStatusId
GROUP BY
  TS.taskStatusName
ORDER BY
  VolunteerCount DESC;
  
-- Answer: 
-- 'closed','4'
-- 'ongoing','2'
-- 'pending','1'


-- 12) Which task creates the heaviest set of packages and what is the weight? Show both the taskCode and the weight (You should be able to do this without using any sub-queries).
-- Q12

SELECT task.taskCode,
  SUM(P.packageWeight) AS TotalWeight
FROM task
  INNER JOIN package AS P
    ON task.taskCode = P.taskCode
GROUP BY task.taskCode
ORDER BY TotalWeight DESC
LIMIT 1;

-- Answer: 
-- 106 
-- 308

-- 13) How many tasks are there that do not have a type of “packing”?
-- Q13

SELECT
  COUNT(task.taskCode)
FROM task 
  INNER JOIN task_type AS TT
    ON task.taskTypeId = TT.taskTypeId
WHERE TT.taskTypeName <> 'packing';

-- 14) Of those items that have been packed, which item (or items) were touched by fewer than 3 volunteers? Answer should be item descriptions.
-- Q14
SELECT item.itemDescription
FROM item 
  INNER JOIN package_contents AS PC
    ON item.itemId = PC.itemId
  INNER JOIN package AS P
    ON PC.packageId = P.packageId
  INNER JOIN task 
    ON P.taskCode = task.taskCode
  INNER JOIN assignment AS assign
    ON task.taskCode = assign.taskCode
GROUP BY item.itemDescription
HAVING COUNT(DISTINCT assign.volunteerId) < 3;

-- Answer: 
-- 'can of spam'
-- 'dried fruit'
-- 'men's coat'
-- 'sleeping bag'
-- 'women's coat'

-- 15) Which packages have a total value of more than 100? Show the packageIds and their value sorted from lowest to highest
-- Q15
SELECT P.packageId,
  SUM(item.itemValue * PC.itemQuantity) AS TotalValue
FROM package AS P
  INNER JOIN package_contents AS PC
    ON P.packageId = PC.packageId
  INNER JOIN item 
    ON PC.itemId = item.itemId
GROUP BY P.packageId
HAVING TotalValue > 100
ORDER BY TotalValue ASC;

-- Answer: 
-- '6','150'
-- '2','151'
-- '4','350'
-- '10','750'
-- '5','1420'
